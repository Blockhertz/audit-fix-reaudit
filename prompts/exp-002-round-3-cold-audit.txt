    self.balances[msg.sender] += msg.value

@external
@nonreentrant("withdraw")
def withdraw(amount: uint256):
    # CHECK
    assert self.balances[msg.sender] >= amount, "Insufficient balance"

    # EFFECT
    self.balances[msg.sender] -= amount

    # INTERACTION
    send(msg.sender, amount)

@external
@view
